** Better Spam
- Add bindings to add/remove from spam
#+begin_src emacs-lisp
  (defun my/gnus-move-to-folder (folder)
    "Move current (or process-marked) articles to FOLDER on this group's server."
    (let ((group (or gnus-newsgroup-name "")))
      (if (string-match "\\`\\(nnimap\\+[^:]+\\):" group)
          (gnus-summary-move-article nil (concat (match-string 1 group) ":" folder))
        (user-error "Not an nnimap group"))))

  (defun my/gnus-mark-spam ()
    "Move article(s) to this account's Spam folder."
    (interactive)
    (my/gnus-move-to-folder "[Gmail]/Spam"))

  (defun my/gnus-mark-not-spam ()
    "Move article(s) out of Spam into this account's INBOX."
    (interactive)
    (my/gnus-move-to-folder "INBOX"))

  (with-eval-after-load 'gnus-sum
    (evil-define-key 'normal gnus-summary-mode-map
      (kbd "s") #'my/gnus-mark-spam
      (kbd "S") #'my/gnus-mark-not-spam))
#+end_src
